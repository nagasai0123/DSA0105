#include<iostream>
using namespace std;
int main(){
  int n=145;
  int temp=n;
  int sum=0;
  while(temp>0){
    int digit=temp%10;
    int fact=1;
    for(int i=1;i<=digit;i++){
      fact*=i;
    }
    sum+=fact;
    temp/=10;
  }
  if(sum==n)
    cout<<n<<"is a strong number.";
  else
    cout<<n<<"is not a strong number.";
  return 0;
}
