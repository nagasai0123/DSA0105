#include <iostream>
using namespace std;
int main() {
    int rows = 2, cols = 3;
    int **a = new int*[rows];
    for(int i = 0; i < rows; i++)
        a[i] = new int[cols];
    a[0][0] = 1;
    a[0][1] = 2;
    a[0][2] = 3;
    a[1][0] = 4;
    a[1][1] = 5;
    a[1][2] = 6;
    for(int i = 0; i < rows; i++) {
        for(int j = 0; j < cols; j++)
            cout << a[i][j] << " ";
        cout << endl;
    }
    for(int i = 0; i < rows; i++)
        delete[] a[i];
    delete[] a;
    return 0;
}
