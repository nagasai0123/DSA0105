#include<iostream>
#include<string>
using namespace std;
int countWords(string str){
  int count=1;
  for(int i=0;i<str.length();i++){
    if(str[i]==' ')
      count++;
  }
  return count;
}
int main(){
  string str="I love c++";
  cout<< "Number of words="<<countWords(str);
  return 0;
}
