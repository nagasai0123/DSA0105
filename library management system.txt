#include <iostream>
using namespace std;

class Library {
    string name[10], author[10];
    bool available[10];
    int count = 0;

public:
    void addBook(string n, string a) {
        name[count] = n;
        author[count] = a;
        available[count] = true;
        count++;
    }

    void displayBooks() {
        cout << "Books in the library:" << endl;

        for(int i = 0; i < count; i++) {
            if(available[i])
                cout << "Book Name: " << name[i] << ", Author: "
                     << author[i] << ", Availability: Available" << endl;
        }
        cout << endl;
    }

    void issueBook(string n) {
        for(int i = 0; i < count; i++) {
            if(name[i] == n)
                available[i] = false;
        }
    }

    void returnBook(string n) {
        for(int i = 0; i < count; i++) {
            if(name[i] == n)
                available[i] = true;
        }
    }
};

int main() {
    Library library;

    library.addBook("The Great Gatsby", "F. Scott Fitzgerald");
    library.addBook("To Kill a Mockingbird", "Harper Lee");

    library.displayBooks();

    library.issueBook("The Great Gatsby");
    library.displayBooks();

    library.returnBook("The Great Gatsby");
    library.displayBooks();

    return 0;
}
