#include<iostream>
using namespace std;
int main(){
  int n=19;
  int temp=n;
  while(temp!=1&&temp!=4){
    int sum=0;
    while(temp>0){
      int digit=temp%10;
      sum+=digit*digit;
      temp/=10;
    }
    temp=sum;
  }
  if(temp==1)
    cout<<n<<"is a happy number.";
  else
    cout<<n<<"is not a happy number.";
  return 0;
}
