#include<iostream>
using namespace std;
void findMinMax(int a[],int n){
  int min=a[0],max=a[0];
  for(int i=1;i<n;i++){
    if(a[i]<min)
      min=a[i];
    if(a[i]>max)
      max=a[i];
  }
  cout<<"Minimum="<<min<<endl;
  cout<<"Maximum="<<max;
}
int main(){
  int a[]={10,20,5,40,30};
  findMinMax(a,5);
  return 0;
}
