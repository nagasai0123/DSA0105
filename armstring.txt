#include<iostream>
using namespace std;
int main(){
  int num=153;
  int original=num;
  int sum=0;
  while (num>0){
    int digit=num%10;
    sum=sum+(digit*digit*digit);
    num=num/10;
  }
  if(sum==original)
    cout<<original<<"is an armstrong number";
  else
    cout<<original<<"is not an armstrong number";
  return 0;
}
