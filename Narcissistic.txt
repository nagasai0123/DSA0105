#include<iostream>
using namespace std;
int main(){
  int n=153;
  int temp=n;
  int sum=0;
  int digits=0;
  while(temp>0){
    digits++;
    temp/=10;
  }
  temp=n;
  while(temp>0){
    int digit=temp%10;
    int power=1;
    for(int i=1;i<=digits;i++){
      power*=digit;
    }
    sum+=power;
    temp/=10;
  }
  if(sum==n)
    cout<<n<<"is a Narcissistic number";
  else
    cout<<n<<"is not a Narcissistic number";
  return 0;
}
