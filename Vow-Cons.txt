#include<iostream>
using namespace std;
int main(){
  char ch;
  cin>>ch;
  ch= tolower(ch);
  if(ch=='a'||ch=='e'||ch=='i'||ch=='o'||ch=='u'){
    cout<<ch<<"is a vowel"<<endl;
  }else if(ch>='a'&&ch=='z'){
    cout<<ch<<"is a consonant"<<endl;
  }else{
    cout<<"invalid input"<<endl;
  }
}
