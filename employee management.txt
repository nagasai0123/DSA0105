#include <iostream>
using namespace std;

class Employee {
    string name, id;
    float salary;

public:
    Employee(string n, string i, float s) {
        name = n;
        id = i;
        salary = s;
    }

    void raiseSalary(float amount) {
        salary += amount;
    }

    void display() {
        cout << "Employee Name: " << name << endl;
        cout << "Employee ID: " << id << endl;
        cout << "Salary: " << salary;
    }
};

int main() {
    Employee e("John Doe", "E12345", 10000);

    e.raiseSalary(5000);
    e.display();

    return 0;
}
